;---------------Escreve String-----------------
escreve_string proc
    push ES         ;salva contextos
    push BX

    mov BX, DS      ;iguala BX a DS
    mov ES, BX      ;ES fica igual a DS

    mov ah,13h      ;atribui funcao 13 da interrupcao
    mov al,01h      ;atribui a al para mover o cursor
    pop BX          ;recupera contexto de BX para obter a cor       
    xor bh,bh       ;video page 0

    int 10h         ;chama interrup????o para escrever a string em questao
    
    pop ES          ;recupera contexto
    ret
endp

;---------------Terreno Procedural (gerador por coluna)-----------------
; Entrada: DX = scroll_offset (0..large)
; Gera uma superf�cie usando duas camadas da tabela `sin_tab` e preenche verticalmente
render_terreno_gen proc
    push ax
    push bx
    push cx
    push dx
    push si
    push di
    push bp
    cmp fase,2
    jne continuar
    mov cl, [terrain_fill2_color]
    mov [terrain_fill_color], cl
    mov cx, [terrain_base2_y]
    mov  [terrain_base_y], cx
    mov cl, [terrain_surface2_color]
    mov [terrain_surface_color], cl
    xor cx, cx
continuar:
    mov ax, 0A000h
    mov es, ax

    mov bp, dx          ; BP = scroll offset (entrada)
    xor bx, bx          ; BX = screen x (0..319)
    mov si, 320         ; SI = columns remaining

.col_loop:
    mov ax, bx
    add ax, bp          ; AX = global_x

    ; idx1 = global_x & 63
    mov cx, ax
    and cx, 63
    mov di, OFFSET sin_tab
    add di, cx
    mov al, [di]

    ; idx2 = (global_x >> 2) & 63
    mov cx, ax
    shr cx, 2
    and cx, 63
    mov di, OFFSET sin_tab
    add di, cx
    mov dl, [di]

    add al, dl
    shr al, 1           ; average normal
    add al, al          ; escalonamento para amplitude vis�vel
    mov ah, 0           ; AX = offset (0..~60)

    ; surface_y = terrain_base_y - AX
    mov cx, [terrain_base_y]
    sub cx, ax          ; CX = surface_y

    ; DI = surface_y * 320 + screen_x
    mov ax, cx
    mov dx, 320
    mul dx               ; DX:AX = AX*320
    mov di, ax
    add di, bx

    ; pixel de superf�cie
    mov al, [terrain_surface_color]
    mov es:[di], al

    ; preenche abaixo com fill_color
    add di, 320
    mov dx, [terrain_base_y]
    sub dx, cx           ; DX = n�mero de linhas abaixo (inclui superf�cie j� escrita)
    mov al, [terrain_fill_color]
    cmp dx, 0
    je .force_blue_fill
.fill_loop:
    mov es:[di], al
    add di, 320
    dec dx
    jnz .fill_loop

.force_blue_fill:
    ; sempre preenche de azul de y=160 at� y=200
    mov ax, 161         ; linha inicial
    mov dx, 320
    mul dx               ; AX = 161 * 320
    add ax, bx           ; AX += coluna atual
    mov di, ax           ; DI = posi��o y=160
    
    mov cx, 40           ; 40 linhas (160 at� 199)
    mov al, [terrain_fill_color]  ; cor azul
.blue_loop:
    mov es:[di], al
    add di, 320
    dec cx
    jnz .blue_loop

    inc bx               ; pr�xima coluna na tela
    dec si
    jnz .col_loop

    pop bp
    pop di
    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    ret
endp

;---------------Converte pontuacao para "00000"-----------------
; Entrada:  usa [pontuacao]
; Sa�da:    escreve 5 d�gitos ASCII em 'score_digits'
score_to_str proc
    push ax
    push bx
    push cx
    push dx
    push si

    mov ax, [pontuacao]
    mov bx, 10
    mov cx, 5
    mov si, OFFSET score_digits
    add si, 4              ; come�a do �ltimo d�gito

conv_loop:
    xor dx, dx
    div bx                 ; AX = AX/10 ; DX = AX%10
    add dl, '0'
    mov [si], dl
    dec si
    loop conv_loop

    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    ret
endp

;---------------Converte segundos para "MM:SS"-----------------
; Usa: tempo_segundos
; Sai: time_digits = MM:SS (2 d�gitos por campo)
tempo_to_str proc
    push ax
    push bx
    push cx
    push dx
    push si
    push di

    ; AX = segundos totais
    mov ax, [tempo_segundos]

    ; divide por 60 -> AX = minutos, DX = segundos
    xor dx, dx
    mov bx, 60
    div bx
    ; AX = minutos, DX = segundos
    ; escreve minutos em time_digits[0..1]
    push dx                ; guarda os segundos
    mov bx, 10
    xor dx, dx
    div bx                 ; AX/10 -> AX=dezenas (AX), DX=unidades
    mov si, OFFSET time_digits
    mov bl, al             ; dezenas em AL -> BL
    add bl, '0'
    mov [si+0], bl
    mov bl, dl             ; unidades em DL -> BL
    add bl, '0'
    mov [si+1], bl
    pop dx                 ; restaura segundos

    ; garante ':'
    mov byte ptr [si+2], ':'

    ; escreve segundos em time_digits[3..4]
    mov ax, dx             ; AX = seg (0..59)
    mov bx, 10
    xor dx, dx
    div bx                 ; AX=dezenas, DX=unidades
    mov bl, al             ; dezenas de segundos
    add bl, '0'
    mov [si+3], bl
    mov bl, dl             ; unidades de segundos
    add bl, '0'
    mov [si+4], bl

    pop di
    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    ret
endp

;---------------Tick do temporizador (60 frames = 1s)-------
timer_tick proc
    push ax
    push bx
    push cx
    push dx
    push si

    ; l� ticks BIOS (CX:DX)
    mov ah, 00h
    int 1Ah
    ; DX = low ticks (increments ~18.2Hz)

    mov si, dx
    ; compara com �ltimo
    mov ax, [bios_last_tick]
    cmp si, ax
    je .no_change

    ; houve um tick novo
    mov [bios_last_tick], si
    ; incrementa acumulador
    mov al, [bios_tick_acc]
    inc al
    mov [bios_tick_acc], al
    cmp al, 18
    jb .no_change

    ; 1 segundo passou (18 ticks)
    mov byte ptr [bios_tick_acc], 0
    ; concede 10 pontos por segundo ativo
    add word ptr [pontuacao], 10
    
    ; atualiza display do score
    call atualiza_score

    mov ax, [tempo_segundos]
    cmp ax, 0
    je .no_change
    dec ax
    mov [tempo_segundos], ax
    call desenha_tempo

.no_change:
    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    ret
endp

;---------------Delay em segundos (BIOS tick 18.2Hz)---------
; Entrada: AL = segundos (0..255)
delay_segundos proc
    push ax
    push bx
    push cx
    push dx
    push si
    push di
    push bp

    ; delta = seconds * 18 (ticks per second)
    xor ah, ah
    mov bl, al
    mov ax, 18
    mul bl            ; DX:AX = 18 * seconds
    mov bx, ax        ; BX = delta_lo (fits in 16-bit)

    ; read start ticks -> CX = start_hi, DX = start_lo
    mov ah, 00h
    int 1Ah
    mov si, cx        ; SI = start_hi
    mov di, dx        ; DI = start_lo

    ; compute target = start + delta
    mov ax, di        ; AX = start_lo
    add ax, bx        ; AX = target_lo (CF set if overflow)
    mov di, ax        ; DI = target_lo
    mov ax, si        ; AX = start_hi
    adc ax, 0         ; AX = target_hi (add carry)
    mov bp, ax        ; BP = target_hi

.wait_loop:
    mov ah, 00h
    int 1Ah            ; CX = cur_hi, DX = cur_lo
    cmp cx, bp
    ja  .done_wait
    jb  .wait_loop
    ; equal high -> compare low
    cmp dx, di
    jae .done_wait
    jmp .wait_loop

.done_wait:
    pop bp
    pop di
    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    ret
endp

; move_vertical

    push ax

    push si

    cmp si, 0     ; calcula linha a apagar em BX
    jne indo_baixo

    add bx, 12         ; linha a apagar = y + (altura-1)

    mov bx, dx
    dec bx             ; linha a apagar = y - 1

    mov ax, bx         ; AX = linha
    mov di, 320

    add ax, cx         ; AX += coluna inicial
    mov di, ax         ; DI = endere�o inicial na VRAM (ES:DI)

    mov cx, 29

    inc di
    dec cx
    jnz clear_loop_v

    pop di
    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    ret
endp

;-------------------Gerar Inimigos------------------------------
gera_inimigo proc
    
    mov DX, gen_time            ;passa intervalo para gerar para DX
    cmp tempo_inimigo, DX     ;ve se ja deu seu tempo para estar na tela 
    jnge ainda_nao_2            ;se nao deu, ignora
    mov BX, max_inimigos      ;se deu, verifica se nao extrapolou o limite de inimigos
    cmp num_inimigos, BX      ;verifica se ja passou do limite de inimigos
    jge ainda_nao_2             ;se sim, ignora
     
    call gera_numero_aleatorio  ;senao, gera numero aleatorio. DX ja fica com a posicao randomica, representando sua posicao y
    inc num_inimigos          ;incrementa o numero de inimigos
    

    mov BX, offset inimigos   ;passa para BX informacoes dos inimigos

    mov AX, max_inimigos      ;passa o numero maximo de inimigos para AX
    push DX                     ;salva DX (que possui o numero aleatorio) pois ira ser perdido na multiplicacao
    mov DX, 6                   ;muda DX para 6 (tamanho do bloco de dados de cada inimigo)
    mul DX                      ;multiplica o maximo de inimigos por 6, descobrindo a posicao que estara quando tiver excedido o limite
    pop DX
    add AX, offset inimigos   ;AX carrega a posicao onde loop deve encerrar, servindo de comparador
   
testes:
    cmp WORD PTR [BX+4], 0      ;verifica se inimigo nessa posicao ja foi gerado
    jne proximo                 ;se sim, vai para o proximo no vetor
    mov WORD PTR [BX], 310      ;passa posicao da extrema direita da tela para posicao x do inimigo
    mov WORD PTR [BX+2], DX     ;passa o numero aleatorio para a posicao y do inimigo
    mov WORD PTR [BX+4], 1      ;liga flag de existencia do inimigo em questao
    mov tempo_inimigo, 0      ;apos criar um novo inimigo, reinicia a contagem do tempo
    jmp ainda_nao_2   
proximo:
    add BX, 6                   ;incrementa posicao do vetor em 6, indo para os dados do proximo inimigo
    cmp BX, AX                  ;verifica se ja chegou no fim do vetor
    jge ainda_nao_2             ;se sim, termina
    jmp testes                  ;senao, faz todo o processo para o novo inimigo

ainda_nao_2:
    ret
endp

;AX recebe x1 e BX o y1
;CX recebe x2 e DX o y2
;retorna em DX 0 se nao ha colisao, 1 se ha
verifica_colisao proc
    push AX             ;salva contexto
    push BX
    push CX

    ;inimigo_x + 29 > nave_x   &&   inimigo_x < nave_x + 29   &&   inimigo_y + 13 > nave_y   &&   inimigo_y < nave_y + 13
    
    add AX, 29          ;largura do inimigo (29)
    cmp AX, CX          ;compara inimigo_x com colisor_x
    jng sem_colisao
    sub AX, 29

    add CX, 29          ;largura do colisor (29)
    cmp AX, CX          ;compara inimigo_x com colisor_x
    jnl sem_colisao
    sub CX, 29
    
    add BX, 13          ;altura do inimigo (13)
    cmp BX, DX          ;compara inimigo_y com colisor_y
    jng sem_colisao
    sub BX, 13
    

    add DX, 13          ;altura do colisor (13)
    cmp BX, DX          ;compara inimigo_y com colisor_y
    jnl sem_colisao
    sub DX, 13
    
    
    mov DX, 1           ;se chegou aqui, houve colisao. Seta flag 
    jmp final_colisao   ;termina
    
sem_colisao:
    xor DX, DX          ;nao houve colisao. Desliga flag
    
final_colisao:

    pop CX          ;recupera contexto
    pop BX
    pop AX
    ret
endp
move_inimigo proc
    push SI                     ;salva contexto
    push DI
    push AX
    push BX
    push CX
    push DX

    mov DI, offset inimigos   ;coloca dados do inimigo em DI

    mul DX                      ;multiplica o maximo de inimigos por 6, descobrindo a posicao que estara quando tiver excedido o limite
    add AX, offset inimigos   ;AX carrega a posicao onde loop deve encerrar, servindo de comparador
    mov SI, AX                  ;comparador vai para SI

    dec WORD PTR [DI]           ;altera a posicao x do inimigo, reduzindo em uma unidade

    pop SI                      ;recupera contexto

    mov BX, [DI+2]              ;passa posicao y para BX

    mov DX, [nave_pos+2]        ;passa posicao y da nave para DX
    call verifica_colisao       ;verifica se houve colisao
    cmp DX, 1                   ;se sim
    je colide_player            ;trata colisao com player
    
    mov CX, [missil_pos]        ;mesmo processo vericando colisao com missil
    mov DX, [missil_pos+2]

    mov BX, offset nave_ini
    call pinta_13X29            ;pinta inimigo com nave_ini e posicoes X e Y em CX e DX

    jmp next                    ;muda para o proximo no vetor

colide_player:
    dec byte ptr [nave_vida]               ;reduz a vida da nave
    call pinta_vida             ;pinta a nova vida
    jmp destroi_inimigo_sem_pontos  ;e destroi o inimigo (sem pontos por colisao)

    jmp destroi_inimigo_com_pontos  ;destroi inimigo com pontos

    mov CX, [DI]             ;apaga o inimigo em questao
    mov DX, [DI+2]
    mov SI, OFFSET nave_ini
    call apagar_sprite_13x29
    mov WORD PTR [DI+4], 0
    dec word ptr [num_inimigos]
    add word ptr [pontuacao], 100  ;adiciona 100 pontos por inimigo destruido

    ; NAO adiciona pontos - inimigo saiu da tela
    call desenha_nave
    jmp next

    ; alias para manter compatibilidade - destroi com pontos
    jmp destroi_inimigo_com_pontos

    cmp DI, SI                  ;verifica se ja chegou no fim do vetor

    pop CX
    pop BX
    pop AX
    pop DI
    pop SI
    ret

gera_missil proc

    push dx

;---------------Mover M�ssil---------------
move_missil proc
    push ax
    push cx
    push dx
    
    cmp byte ptr [missil_exists], 0  ;verifica se missil existe
    je .fim_move                     ;se nao existe, sai
    
    ; apagar pixel atual
    mov CX, [missil_pos]
    mov DX, [missil_pos+2]
    mov AL, 0                        ;cor preta (apagar)
    call pinta_pixel
    
    ; mover para direita
    mov CX, [missil_pos]
    add CX, 3                        ;velocidade do missil
    cmp CX, 319                      ;verifica limite da tela
    jg .destroy_missil               ;se passou do limite, destroi
    
    ; atualizar posicao
    mov [missil_pos], CX
    
    ; desenhar na nova posicao
    mov DX, [missil_pos+2]
    mov AL, 15                       ;cor branca
    call pinta_pixel
    jmp .fim_move
    
.destroy_missil:
    mov byte ptr [missil_exists], 0  ;desativa missil
    
.fim_move:
    pop dx
    pop cx
    pop ax
    ret
endp

;---------------Verifica Colisao com Terreno---------------
verifica_colisao_terreno proc
    push ax
    push bx
    push cx
    push dx
    push si
    push di
    
    ; calcula a posi��o Y do terreno na posi��o X da nave
    mov bx, [nave_pos]      ; x da nave
    
    ; usa mesma l�gica do render_terreno_gen para calcular altura do terreno
    mov ax, bx              ; global_x = nave_pos_x
    
    ; idx1 = global_x & 63
    mov cx, ax
    and cx, 63
    mov di, OFFSET sin_tab
    add di, cx
    mov al, [di]

    ; idx2 = (global_x >> 2) & 63
    mov cx, ax
    shr cx, 2
    and cx, 63
    mov di, OFFSET sin_tab
    add di, cx
    mov dl, [di]

    add al, dl
    shr al, 1           ; average normal
    add al, al          ; escalonamento para amplitude vis�vel
    mov ah, 0           ; AX = offset (0..~60)

    ; surface_y = terrain_base_y - AX
    mov cx, [terrain_base_y]
    sub cx, ax          ; CX = altura do terreno nesta posi��o
    sub cx, 2
    ; verifica se a parte inferior da nave (nave_y + 13) toca o terreno
    mov ax, [nave_pos+2]    ; y da nave
    add ax, 13              ; + altura da nave
    cmp ax, cx              ; compara com altura do terreno
    jl .sem_colisao         ; se nave est� acima do terreno, ok
    
    ; colidiu com terreno - perde vida
    dec byte ptr [nave_vida]
    call pinta_vida
    
    ; reposiciona nave acima do terreno com margem de seguran�a
    sub cx, 15              ; margem de seguran�a
    mov [nave_pos+2], cx
    
.sem_colisao:
    pop di
    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    ret
endp

;-------------------Verificar Tecla Jogo----------------------------

verifica_tecla_jogo proc
    push BX
    
    mov ah, 1h              ;configura interrupcao
    int 16h                 ;verifica tecla no buffer
    jz nao_jogada1           ;se nao tiver, ignora
    xor ah,ah               ;se houver, muda a configuracao e
    int 16h                 ;pega a tecla no buffer
    cmp ah, 50h             ;se for down arrow
    jne not_down            ;senao, verifica pra ver se e cima
    cmp [nave_pos+2], 170   ;se sim, verifica se a nave esta no limite inferior da tela
    je nao_jogada1           ;se ja esta no limite da tela, nao faz nada
    mov CX, [nave_pos]      ;nave_x para CX
    mov DX, [nave_pos+2]    ;nave_y para DX
    mov SI, 1               ;dire��o baixo (1)
    call move_vertical      ;apaga linha superior
    inc [nave_pos+2]        ;incrementa uma posicao 
    call verifica_colisao_terreno  ;verifica se colidiu com terreno
    call desenha_nave
    jmp nao_jogada          ;pula para o final
not_down: 
    cmp ah, 48h             ;se for up arrow
    jne not_up              ;se nao for nem baixo nem cima, verifica outras teclas
    cmp [nave_pos+2],0      ;se for, verifica se nave esta no limite superior da tela
    je nao_jogada           ;se ja esta no limite da tela, nao faz nada
    mov CX, [nave_pos]      ;nave_x para CX
    mov DX, [nave_pos+2]    ;nave_y para DX
    mov SI, 0               ;dire��o cima (0)
    call move_vertical      ;apaga linha inferior
    dec [nave_pos+2]        ;decrementa posicao y da nave
    call verifica_colisao_terreno  ;verifica se colidiu com terreno
    call desenha_nave
    jmp nao_jogada          ;pula para o final
nao_jogada1:
    jmp nao_jogada
not_up:
    cmp ah, 4Bh             ;se for left arrow
    jne not_left            ;se nao for esquerda, verifica direita
    cmp [nave_pos], 0       ;verifica se a nave esta no limite esquerdo da tela
    je nao_jogada           ;se ja esta no limite da tela, nao faz nada
    mov CX, [nave_pos]      ;nave_x para CX
    mov DX, [nave_pos+2]    ;nave_y para DX
    mov SI, 0               ;direcao esquerda (0)
    call move_horizontal    ;apaga coluna da direita
    dec [nave_pos]          ;decrementa posicao x da nave
    call verifica_colisao_terreno  ;verifica se colidiu com terreno
    call desenha_nave       ;redesenha a nave
    jmp nao_jogada          ;pula para o final
not_left:
    cmp ah, 4Dh             ;se for right arrow
    jne tiro                ;se nao for direita, verifica spacebar
    cmp [nave_pos], 291     ;verifica se a nave esta no limite direito (320-29=291)
    je nao_jogada           ;se ja esta no limite da tela, nao faz nada
    mov CX, [nave_pos]      ;nave_x para CX
    mov DX, [nave_pos+2]    ;nave_y para DX
    mov SI, 1               ;direcao direita (1)
    call move_horizontal    ;apaga coluna da esquerda
    inc [nave_pos]          ;incrementa posicao x da nave
    call verifica_colisao_terreno  ;verifica se colidiu com terreno
    call desenha_nave       ;redesenha a nave
    jmp nao_jogada          ;pula para o final
tiro:
    cmp ah, 39h             ;spacebar
    jne nao_jogada          ;se nao for, ignora
    call gera_missil        ;se for, gera missil      
nao_jogada:    
    pop BX
    ret
endp

    ; testa se h� tecla dispon�vel
    mov ah, 01h

    ; l� a tecla (bloqueante)

    cmp al, 13

    ; decrementa e trava em 0
    mov ax, [opcao_menu]

    call altera_opcao
    jmp .sem_tecla

    ; incrementa e trava em 1
    mov ax, [opcao_menu]
    cmp ax, 1
    je .sem_tecla         ; j� no fim
    inc ax
    mov [opcao_menu], ax
    call altera_opcao
    jmp .sem_tecla

.enter_pres:
    mov [flag_entrada], 1
    ret

.sem_tecla:
    ret
endp

INICIO:
    INICIO:
    MOV AX, @data
    MOV DS, AX
    MOV AX, 0A000h
    mov ES, AX

    ; gerar posi��o X aleat�ria
    call gerar_posicao_inimigo

    call inicia_video

    ; DESENHA O MENU COMPLETO AQUI
    call desenha_menu_inicial

menu_loop:

    call movimento_tela_inicial

    mov cx, 30000
delay_lp:
    loop delay_lp

    call verifica_tecla_menu

    cmp flag_entrada, 1
    jne .apenas_mover

    mov flag_entrada, 0

    cmp opcao_menu, 0
    je jogo

    cmp opcao_menu, 1
    je sair
    jmp menu_loop

.apenas_mover:
    call altera_opcao
    jmp menu_loop

jogo:
    inc fase
    call mudar_tela
    cmp fase,1
    jne desenha_2
    call desenha_fase1
    jmp continua
desenha_2:
    call desenha_fase2
continua:
    ; espera 4 segundos na tela da fase
    mov al, 4
    call delay_segundos

    ; volta para tela de jogo limpa e desenha barra de status
    call mudar_tela
    call comeca_fase
    call desenha_status_bar

loop_jogo:
    ; checa ESC para sair
    call verifica_tecla_jogo
    mov ah, 01h
    int 16h
    jz .no_key2
    mov ah, 00h
    int 16h
    cmp al, 27
    je sair
.no_key2:

    ; atraso de frame simples
    mov cx, 30000
._delay:
    loop ._delay

    ; execuções durante o jogo
    call durante_jogo

    ; verifica se perdeu todas as vidas
    cmp byte ptr [nave_vida], 0
    je sair

    ; quando tempo chegar a zero, encerra
    mov ax, [tempo_segundos]
    cmp ax, 0
    jne loop_jogo
    cmp fase,4
    jne jogo
    jmp fim

sair:
    call mudar_tela
    call desenha_game_over
    jmp fim

fim:
end INICIO
