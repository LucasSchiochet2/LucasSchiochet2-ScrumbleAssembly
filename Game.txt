;---------------Escreve String-----------------
escreve_string proc
    push ES         ;salva contextos
    push BX

    mov BX, DS      ;iguala BX a DS
    mov ES, BX      ;ES fica igual a DS

    mov ah,13h      ;atribui funcao 13 da interrupcao
    mov al,01h      ;atribui a al para mover o cursor
    pop BX          ;recupera contexto de BX para obter a cor       
    xor bh,bh       ;video page 0

    int 10h         ;chama interrup????o para escrever a string em questao
    
    pop ES          ;recupera contexto
    ret
endp
;---------------Fim Escreve String-----------------

;------------------Pinta Pixel--------------------
pinta_pixel proc
    push BX
    push DX
    push AX

    mov BX, 320
    mov AX, DX
    mul BX
    mov BX, AX

    pop AX
    pop DX

    add BX, CX
    mov SI, BX

    mov ES:[SI], AL

    pop BX
    ret
endp
;---------------Fim Pinta Pixel--------------------

;---------------Pinta e Apaga Desenho--------------------
pinta_13X29 proc
    push ax
    push bx
    push cx
    push dx
    push si
    push di

    mov si, bx
    mov ax, dx
    mov bx, 320
    mul bx
    add ax, cx
    mov di, ax

    mov dx, 13

linha_loop:
    mov cx, 29

coluna_loop:
    mov al, [si]
    mov es:[di], al
    inc si
    inc di
    loop coluna_loop

    add di, 320-29
    dec dx
    jnz linha_loop

    pop di
    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    ret
endp
;------------------Apaga Desenho----------------------
apagar_sprite_13x29 proc
    ; recebe:
    ;   CX = x
    ;   DX = y
    ;   SI = OFFSET sprite (só para manter padrão, mas não usado)

    push ax
    push bx
    push cx
    push dx
    push di

    mov ax, dx
    mov bx, 320
    mul bx
    add ax, cx
    mov di, ax            ; início do sprite na VRAM

    mov bx, 13            ; altura
linha_apagar:
    mov cx, 29            ; largura
    xor al, al            ; preto
apaga_coluna:
    mov es:[di], al
    inc di
    loop apaga_coluna

    add di, 320 - 29      ; próxima linha
    dec bx
    jnz linha_apagar

    pop di
    pop dx
    pop cx
    pop bx
    pop ax
    ret
endp

desenha_nave proc
    mov CX, nave_pos
    mov DX, [nave_pos+2]
    call pinta_13X29
    ret
endp
;---------------Fim Pinta e Apaga Desenho--------------------

;---------------Muda de Tela--------------------
mudar_tela proc
   
    mov CX, 57600       ;seleciona valor limite na tela
pinta_preto:
    mov SI, CX          ;passa valor do contador para SI
    mov WORD PTR ES:[SI], 0      ;pinta de preto
    loop pinta_preto    ;decrementa contador, verifica se chegou a zero. Senao, continua no loop
    
    ret
endp
;---------------Fim Muda de Tela--------------------

;---------------Movimentação------------------------
; move_horizontal
; - escreve 13 pixels verticalmente com MOV/ADD (sem multiplicações por pixel)
; Entrada:
;   CX = x atual
;   DX = y atual
;   SI = direção (0 = esquerda, 1 = direita)
move_horizontal proc
    push ax
    push bx
    push cx
    push dx
    push si
    push di

    cmp si, 0     ; calcula coluna a apagar em BX
    jne indo_direita

    ; indo para a esquerda -> apagar lado direito do sprite
    mov bx, cx
    add bx, 28         ; coluna a apagar = x + (largura-1)
    jmp prepara

indo_direita:
    mov bx, cx
    dec bx             ; coluna a apagar = x - 1

prepara:
    ; DI_final = DX * 320 + BX
    mov ax, dx         ; AX = y
    mov di, 320
    mul di             ; AX = y * 320
    add ax, bx         ; AX += coluna
    mov di, ax         ; DI = endereço inicial na VRAM (ES:DI)

    ; limpa verticalmente 13 pixels (escreve AL=0 e soma 320 para próxima linha)
    mov cx, 13
    xor al, al

clear_loop:
    mov es:[di], al
    add di, 320
    dec cx
    jnz clear_loop

    pop di
    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    ret
endp
;------------------Fim Movimentação------------------------


;-------------------Incio Video-----------------------------
inicia_video proc
    xor AH, AH
    mov AL, 13h
    int 10h
    ret
endp
;-------------------Fim Incio Video-----------------------------

;-------------------Movimentação na Tela Incial-----------------
movimento_tela_inicial proc
    push ax
    push bx
    push cx
    push dx
    push si
    push di
; 1) NAVE ALIADA
    ; apagar coluna antiga (direção = direita)
    mov cx, [nave_x]
    mov dx, [nave_y]
    mov si, 1              ; 1 = direita (move_horizontal usa SI como direção)
    call move_horizontal

    ; atualizar X
    mov ax, [nave_x]
    inc ax
    cmp ax, 320
    jl .ali_ok

    ; apagar sprite inteiro ANTES de resetar posição
    mov cx, [nave_x]
    mov dx, [nave_y]
    mov si, OFFSET nave
    call apagar_sprite_13x29
    mov ax, 0     ; reaparece à esquerda
.ali_ok:
    mov [nave_x], ax
    ; desenhar nova posição
    mov bx, offset nave
    mov cx, [nave_x]
    mov dx, [nave_y]
    call pinta_13X29
; 2) METEORO
    ; apagar coluna antiga (direção = esquerda)
    mov cx, [meteoro_x]
    mov dx, [meteoro_y]
    mov si, 0              ; 0 = esquerda
    call move_horizontal

    mov ax, [meteoro_x]    ; atualizar X
    dec ax
    cmp ax, 0
    jg .met_ok

    mov cx, [meteoro_x]
    mov dx, [meteoro_y]
    mov si, OFFSET meteoro
    call apagar_sprite_13x29
    mov ax, 320          ; reaparece à direita
.met_ok:
    mov [meteoro_x], ax
    ; desenhar nova posição
    mov bx, offset meteoro
    mov cx, [meteoro_x]
    mov dx, [meteoro_y]
    call pinta_13X29
; 3) inimigo
    ; apagar coluna antiga (usa inimigo_dir)
    mov cx, [inimigo_x]
    mov dx, [inimigo_y]
    mov si, [inimigo_dir]    ; SI = 0 (esq) ou 1 (dir)
    call move_horizontal

    ; atualizar X com ping-pong
    mov ax, [inimigo_x]
    mov bx, [inimigo_dir]    ; bx = direção
    cmp bx, 0
    jne .inimigo_dir_right

    ; indo para esquerda
    dec ax
    cmp ax, 0
    jg .inimigo_depois
    ; bateu na esquerda -> inverter para direita
    mov word ptr [inimigo_dir], 1
    jmp .inimigo_depois
.inimigo_dir_right:
    ; indo para direita
    inc ax
    cmp ax, 291            ; limite (320 - sprite_width(29)) = 291
    jle .inimigo_depois
    ; bateu na direita -> inverter para esquerda
    mov word ptr [inimigo_dir], 0
.inimigo_depois:
    mov [inimigo_x], ax
    ; desenhar inimigo
    mov bx, offset nave_ini
    mov cx, [inimigo_x]
    mov dx, [inimigo_y]
    call pinta_13X29
    pop di
    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    ret
endp

;-------------------Verificar Tecla Menu----------------------------
verifica_tecla_menu proc
    ; testa se há tecla disponível
    mov ah, 01h
    int 16h
    jz .sem_tecla

    ; lê a tecla (bloqueante)
    mov ah, 00h
    int 16h

    ; ENTER?
    cmp al, 13
    je .enter_pres

    ; verificar setas (usa AH)
    cmp ah, 48h        ; seta ↑
    je .seta_cima

    cmp ah, 50h        ; seta ↓
    je .seta_baixo

    jmp .sem_tecla

.seta_cima:
    ; decrementa e trava em 0
    mov ax, [opcao_menu]
    cmp ax, 0
    je .sem_tecla         ; já no topo
    dec ax
    mov [opcao_menu], ax
    call altera_opcao
    jmp .sem_tecla

.seta_baixo:
    ; incrementa e trava em 1
    mov ax, [opcao_menu]
    cmp ax, 1
    je .sem_tecla         ; já no fim
    inc ax
    mov [opcao_menu], ax
    call altera_opcao
    jmp .sem_tecla

.enter_pres:
    mov [flag_entrada], 1
    ret

.sem_tecla:
    ret
endp

INICIO:

    MOV AX, @data
    MOV DS, AX

    mov ES, AX

    mov ax, seed
    add ax, 37

    div bx

menu_loop:

    call movimento_tela_inicial

    mov cx, 30000
delay_lp:
    loop delay_lp

    call verifica_tecla_menu

    cmp flag_entrada, 1
    jne .apenas_mover

    mov flag_entrada, 0

    cmp opcao_menu, 0
    je jogo

    cmp opcao_menu, 1
    je sair
    jmp menu_loop

.apenas_mover:
    call altera_opcao
    jmp menu_loop

    call mudar_tela
    call desenha_fase1
    jmp fim

    jmp fim

fim:
