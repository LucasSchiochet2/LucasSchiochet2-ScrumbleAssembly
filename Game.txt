;---------------------------------------------------------
; Funcao: Escreve uma string na tela usando a BIOS (INT 10h)
; Parametros de entrada: 
;   BP = Offset da string
;   CX = Tamanho da string
;   DH = Linha
;   DL = Coluna
;   BL = Atributo de cor
; Parametros de saida: Nenhum
;---------------------------------------------------------
escreve_string proc
    push ES         ;salva contextos
    push BX

    mov BX, DS      ;iguala BX a DS
    mov ES, BX      ;ES fica igual a DS

    mov ah,13h      ;atribui funcao 13 da interrupcao
    mov al,01h      ;atribui a al para mover o cursor
    pop BX          ;recupera contexto de BX para obter a cor       
    xor bh,bh       ;video page 0

    int 10h         ;chama interrupcao para escrever a string em questao
    
    pop ES          ;recupera contexto
    ret
endp
;---------------Fim Escreve String-----------------
;---------------------------------------------------------
; Funcao: Verifica se a fase atual e a 2 e ajusta as cores e tabelas do terreno
; Parametros de entrada: [fase]
; Parametros de saida: Altera [terrain_fill_color], [terrain_base_y], [terrain_surface_color], [sin_tab] se fase == 2
;---------------------------------------------------------
verificar_fase2 proc
    push ax
    push bx
    push cx
    push dx
    push si
    push di
    push bp
    
    cmp fase, 2
    jne continuar
    
    ; ajustar cores e altura do terreno para fase 2
    mov cl, [terrain_fill2_color]
    mov [terrain_fill_color], cl
    mov cx, [terrain_base2_y]
    mov [terrain_base_y], cx
    mov cl, [terrain_surface2_color]
    mov [terrain_surface_color], cl
    
    ; copiar sin_tab2 para sin_tab (sobrescrever)
    push ds
    push es
    mov ax, @data
    mov ds, ax
    mov es, ax
    mov si, OFFSET sin_tab2
    mov di, OFFSET sin_tab
    mov cx, 64
.copy_loop:
    mov al, [si]
    mov [di], al
    inc si
    inc di
    dec cx
    jnz .copy_loop
    pop es
    pop ds
    
continuar:
    pop bp
    pop di
    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    ret
endp

    push di
    push bp
 
    mov ax, 0A000h
    mov es, ax
    call verificar_fase2 
    mov bp, dx          ; BP = scroll offset (entrada)
    xor bx, bx          ; BX = screen x (0..319)
    mov si, 320         ; SI = columns remaining

.col_loop:
    mov ax, bx
    add ax, bp          ; AX = global_x

    ; idx1 = global_x & 63
    mov cx, ax
    and cx, 63
    mov di, OFFSET sin_tab
    add di, cx
    mov al, [di]

    ; idx2 = (global_x >> 2) & 63
    mov cx, ax
    shr cx, 2
    and cx, 63
    mov di, OFFSET sin_tab
    add di, cx
    mov dl, [di]

    add al, dl
    shr al, 1           ; average normal
    add al, al          ; escalonamento para amplitude visivel
    mov ah, 0           ; AX = offset (0..~60)

    ; surface_y = terrain_base_y - AX
    mov cx, [terrain_base_y]
    sub cx, ax          ; CX = surface_y

    ; DI = surface_y * 320 + screen_x
    mov ax, cx
    mov dx, 320
    mul dx               ; DX:AX = AX*320
    mov di, ax
    add di, bx

    ; pixel de superficie
    mov al, [terrain_surface_color]
    mov es:[di], al

    ; preenche abaixo com fill_color
    add di, 320
    mov dx, [terrain_base_y]
    sub dx, cx           ; DX = numero de linhas abaixo (inclui superficie ja escrita)
    mov al, [terrain_fill_color]
    cmp dx, 0
    je .force_blue_fill
.fill_loop:
    mov es:[di], al
    add di, 320
    dec dx
    jnz .fill_loop

.force_blue_fill:
    ; sempre preenche de azul de y=160 ate y=200
    mov ax, 161         ; linha inicial
    mov dx, 320
    mul dx               ; AX = 161 * 320
    add ax, bx           ; AX += coluna atual
    mov di, ax           ; DI = posicao y=160
    
    mov cx, 40           ; 40 linhas (160 ate 199)
    mov al, [terrain_fill_color]  ; cor azul
.blue_loop:
    mov es:[di], al
    add di, 320
    dec cx
    jnz .blue_loop

    inc bx               ; proxima coluna na tela
    dec si
    jnz .col_loop

    pop bp
    pop di
    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    ret
endp

;---------------------------------------------------------
; Funcao: Pinta um bloco de 24x16 pixels (Fase 3)
; Parametros: 
;   BX = Offset do Sprite (bloco3_1 ou bloco3_2)
;   CX = Posicao X
;   DX = Posicao Y (Canto superior esquerdo do bloco)
;---------------------------------------------------------
pinta_bloco_fase3 proc
    push ax
    push bx
    push cx
    push dx
    push si
    push di

    mov si, bx          ; SI aponta para os dados do sprite
    
    ; 1. Calcular endereco inicial de video: DI = (Y * 320) + X
    mov ax, dx
    mov bx, 320
    mul bx              ; DX:AX = Y * 320
    add ax, cx          ; AX = AX + X
    mov di, ax          ; ES:DI aponta para a tela
    
    ; 2. Definir Altura Fixa
    mov dx, 16          ; 16 pixels para qualquer bloco!

    ; 3. Loop de Desenho
.prox_linha:
    mov cx, 24          ; Largura fixa (24 pixels)
    
.prox_pixel:
    mov al, [si]        ; Le a cor do sprite
    
    ; Opcional: Transparencia para cor 0 (preto)
    cmp al, 0
    je .pula_pintura
    mov es:[di], al     ; Pinta na tela
.pula_pintura:

    inc si              ; Proximo byte do sprite
    inc di              ; Proximo pixel da tela
    loop .prox_pixel    ; Repete 24 vezes
    
    ; Fim da linha, vai para a de baixo
    add di, 320 - 24    ; Avanca DI para a proxima linha
    dec dx              ; Decrementa altura
    jnz .prox_linha     ; Se nao acabou a altura, repete

    pop di
    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    ret
pinta_bloco_fase3 endp


;---------------------------------------------------------
; Funcao: Renderiza o terreno da Fase 3 (Blocos)
;---------------------------------------------------------
;---------------------------------------------------------
; Funcao: Renderiza o terreno da Fase 3 (Blocos de Tijolos)
;---------------------------------------------------------
render_terreno_fase3 proc
    push ax
    push bx
    push cx
    push dx
    push si
    push di
    
    mov ax, 0A000h
    mov es, ax

    ; 1. Limpa o fundo (Linha 100 a 200)
    mov di, 32000
    xor ax, ax
    mov cx, 16000
    cld
    rep stosw

    ; 2. Loop pelas colunas do mapa
    mov si, 0           ; Indice da coluna (0 a 13)
    
.loop_colunas:
    ; Ler altura da coluna no mapa
    mov bx, offset mapa_fase3_blocos
    add bx, si
    xor ax, ax
    mov al, [bx]        ; AL = Quantidade de blocos
    
    cmp al, 0
    je .proxima_coluna_mapa
    
    ; Calcular Posicao X = Indice * 24
    push ax
    mov ax, si
    mov cl, 24
    mul cl
    mov cx, ax          ; CX = Coordenada X
    pop ax              ; Recupera quantidade de blocos (AL)
    
    ; Definir Y base (O chao da tela)
    mov dx, 200         
    
    mov ah, al          ; AH = Contador de blocos
    
.loop_empilha:
    push ax             ; Salva regs
    push bx
    push cx
    push dx
    
    ; Decidir qual bloco desenhar
    cmp ah, 1           ; ultimo bloco (o mais alto)
    je .desenha_topo
    
    ; --- Desenha CORPO (bloco3_1) ---
    sub dx, 16          ; Sobe 16 pixels
    mov bx, OFFSET bloco3_1
    call pinta_bloco_fase3
    jmp .pos_desenho_bloco

.desenha_topo:
    ; --- Desenha TOPO (bloco3_2) ---
    sub dx, 16          ; AGORA SOBE 16 (Antes era 7)
    mov bx, OFFSET bloco3_2
    call pinta_bloco_fase3

.pos_desenho_bloco:
    ; Atualizar Y para o proximo loop (o bloco de cima)
    pop dx              ; Recupera Y anterior (190)
    sub dx, 16          ; Sobe 16 pixels para a proxima iteracao
    
    pop cx              ; Recupera X
    pop bx              ; Recupera ponteiro
    pop ax              ; Recupera contadores
    
    dec ah              ; Decrementa blocos restantes
    jnz .loop_empilha
    
.proxima_coluna_mapa:
    inc si
    cmp si, 13          ; Maximo de colunas
    jl .loop_colunas

    pop di
    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    ret
render_terreno_fase3 endp


;---------------------------------------------------------
; Funcao: Converte o valor numerico da pontuacao para string formatada "00000"
; Parametros de entrada: [pontuacao]
; Parametros de saida: Atualiza buffer 'score_digits' com os caracteres ASCII
;---------------------------------------------------------
score_to_str proc
    push ax
    push bx
    push cx
    push dx
    push si

    mov ax, [pontuacao]
    mov bx, 10
    mov cx, 5
    mov si, OFFSET score_digits
    add si, 4              ; comeca do ultimo digito

conv_loop:
    xor dx, dx
    div bx                 ; AX = AX/10 ; DX = AX%10
    add dl, '0'
    mov [si], dl
    dec si
    loop conv_loop

    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    ret
endp

;---------------Converte segundos para "MM:SS"-----------------
; Funcao: Converte o tempo em segundos para string formatada "MM:SS"
; Parametros de entrada: [tempo_segundos]
; Parametros de saida: Atualiza buffer 'time_digits'
;---------------------------------------------------------
tempo_to_str proc
    push ax
    push bx
    push cx
    push dx
    push si
    push di

    ; AX = segundos totais
    mov ax, [tempo_segundos]

    ; divide por 60 -> AX = minutos, DX = segundos
    xor dx, dx
    mov bx, 60
    div bx
    ; AX = minutos, DX = segundos
    ; escreve minutos em time_digits[0..1]
    push dx                ; guarda os segundos
    mov bx, 10
    xor dx, dx
    div bx                 ; AX/10 -> AX=dezenas (AX), DX=unidades
    mov si, OFFSET time_digits
    mov bl, al             ; dezenas em AL -> BL
    add bl, '0'
    mov [si+0], bl
    mov bl, dl             ; unidades em DL -> BL
    add bl, '0'
    mov [si+1], bl
    pop dx                 ; restaura segundos

    ; garante ':'
    mov byte ptr [si+2], ':'

    ; escreve segundos em time_digits[3..4]
    mov ax, dx             ; AX = seg (0..59)
    mov bx, 10
    xor dx, dx
    div bx                 ; AX=dezenas, DX=unidades
    mov bl, al             ; dezenas de segundos
    add bl, '0'
    mov [si+3], bl
    mov bl, dl             ; unidades de segundos
    add bl, '0'
    mov [si+4], bl

    pop di
    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    ret
endp

;---------------------------------------------------------
; Funcao: Gerencia a contagem de tempo baseada no clock da BIOS
; Parametros de entrada: Nenhum (le INT 1Ah)
; Parametros de saida: Atualiza [pontuacao] e [tempo_segundos] a cada segundo
;---------------------------------------------------------
timer_tick proc
    push ax
    push bx
    push cx
    push dx
    push si

    ; la ticks BIOS (CX:DX)
    mov ah, 00h
    int 1Ah
    ; DX = low ticks (increments ~18.2Hz)

    mov si, dx
    ; compara com ultimo
    mov ax, [bios_last_tick]
    cmp si, ax
    je .no_change

    ; houve um tick novo
    mov [bios_last_tick], si
    ; incrementa acumulador
    mov al, [bios_tick_acc]
    inc al
    mov [bios_tick_acc], al
    cmp al, 18
    jb .no_change

    ; 1 segundo passou (18 ticks)
    mov byte ptr [bios_tick_acc], 0
    ; Verifica qual fase e para dar os pontos corretos
    cmp fase, 1
    je .pontos_fase1
    
    cmp fase, 2
    je .pontos_fase2
    
    ; Fase 3
    add word ptr [pontuacao], 20
    jmp .atualiza_tela

.pontos_fase1:
    add word ptr [pontuacao], 10
    jmp .atualiza_tela

.pontos_fase2:
    add word ptr [pontuacao], 15
    
.atualiza_tela:
    
    ; atualiza display do score
    call atualiza_score

    mov ax, [tempo_segundos]
    cmp ax, 0
    je .no_change
    dec ax
    mov [tempo_segundos], ax
    call desenha_tempo

.no_change:
    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    ret
endp

;---------------Desenha vidas (3 icones no centro)-----------
; Funcao: Desenha os icones representativos das vidas restantes
; Parametros de entrada: [vidas], [vidas_start_x]
; Parametros de saida: Desenha os icones na barra de status
;---------------------------------------------------------
desenha_vidas proc
    push ax
    push bx
    push cx
    push dx
    push si
    push di

    ; primeiro limpa toda a area das vidas (3 icones maximos)
    mov si, [vidas_start_x]  ; posicao inicial
    mov dx, 2                ; linha y
    mov bx, 3                ; limpar espaco para 3 vidas

.clear_loop:
    cmp bx, 0
    je .start_draw
    push bx
    push si
    
    ; limpa area de uma vida (24x9 pixels - area maior para garantir limpeza completa)
    mov ax, dx               ; y em AX
    mov di, 320
    mul di                   ; AX = y * 320
    add ax, si               ; AX += x
    mov di, ax               ; DI = posicao
    
    mov ax, 9                ; altura ampliada para garantir limpeza completa
.clear_linha:
    push ax
    mov cx, 24               ; largura ampliada (19 + margem de seguranca)
    xor al, al               ; cor preta
.clear_coluna:
    mov es:[di], al
    inc di
    loop .clear_coluna
    
    add di, 320-24           ; proxima linha (ajustado para nova largura)
    pop ax
    dec ax
    jnz .clear_linha
    
    pop si
    pop bx
    add si, 24               ; proxima posicao (largura + gap)
    dec bx
    jmp .clear_loop

.start_draw:
    ; agora desenha apenas as vidas restantes
    mov ax, [vidas_start_x]
    mov si, ax        ; start_x
    mov dx, 2         ; y
    mov cl, [vidas]   ; desenhar somente vidas ativas
    mov cx, cx        ; extend CL to CX for loop

    cmp cx, 0
    je .fim
    push cx
    mov bx, OFFSET nave_menu
    mov cx, si       ; x em CX (pinta_7x19 espera CX=x)
    mov dx, 2
    call pinta_7x19
    pop cx

    ; avanca X: largura 19 + gap 5 = 24
    add si, 24
    dec cx
    jmp .draw_loop_fixed

.fim:
    pop di
    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    ret
endp

;---------------Delay em segundos-------------------------
; Funcao: Gera um atraso (delay) especificado em segundos
; Parametros de entrada: AL = Segundos a aguardar
; Parametros de saida: Nenhum (pausa a execucao)
;---------------------------------------------------------
delay_segundos proc
    push ax
    push bx
    push cx
    push dx
    push si
    push di
    push bp

    ; delta = seconds * 18 (ticks per second)
    xor ah, ah
    mov bl, al
    mov ax, 18
    mul bl            ; DX:AX = 18 * seconds
    mov bx, ax        ; BX = delta_lo (fits in 16-bit)

    ; read start ticks -> CX = start_hi, DX = start_lo
    mov ah, 00h
    int 1Ah
    mov si, cx        ; SI = start_hi
    mov di, dx        ; DI = start_lo

    ; compute target = start + delta
    mov ax, di        ; AX = start_lo
    add ax, bx        ; AX = target_lo (CF set if overflow)
    mov di, ax        ; DI = target_lo
    mov ax, si        ; AX = start_hi
    adc ax, 0         ; AX = target_hi (add carry)
    mov bp, ax        ; BP = target_hi

.wait_loop:
    mov ah, 00h
    int 1Ah            ; CX = cur_hi, DX = cur_lo
    cmp cx, bp
    ja  .done_wait
    jb  .wait_loop
    ; equal high -> compare low
    cmp dx, di
    jae .done_wait
    jmp .wait_loop

.done_wait:
    pop bp
    pop di
    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    ret
endp

;---------------------------------------------------------
; Funcao: Apaga o rastro vertical de um sprite ao mover (apaga linha superior ou inferior)
; Parametros de entrada: 
;   CX = Posicao X atual
;   DX = Posicao Y atual
;   SI = Direcao (0 = cima, 1 = baixo)
; Parametros de saida: Limpa linha de pixels
;---------------------------------------------------------
move_vertical proc
    push ax
    push bx
    push cx
    push dx
    push si
    push di

    cmp si, 0     ; calcula linha a apagar em BX
    jne indo_baixo

    ; indo para cima -> apagar linha inferior do sprite
    mov bx, dx
    add bx, 12         ; linha a apagar = y + (altura-1)
    jmp prepara_v

indo_baixo:
    mov bx, dx
    dec bx             ; linha a apagar = y - 1

prepara_v:
    ; DI_final = BX * 320 + CX
    mov ax, bx         ; AX = linha
    mov di, 320
    mul di             ; AX = linha * 320
    add ax, cx         ; AX += coluna inicial
    mov di, ax         ; DI = endereco inicial)

    ; limpa horizontalmente 29 pixels (largura do sprite)
    mov cx, 29
    xor al, al

clear_loop_v:
    mov es:[di], al
    inc di
    dec cx
    jnz clear_loop_v

    pop di
    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    ret
endp

;---------------------------------------------------------
; Funcao: Apaga o rastro horizontal de um sprite ao mover (apaga coluna esquerda ou direita)
; Parametros de entrada: 
;   CX = Posicao X atual
;   DX = Posicao Y atual
;   SI = Direcao (0 = esquerda, 1 = direita)
; Parametros de saida: Limpa coluna de pixels
;---------------------------------------------------------
move_horizontal proc
    push ax
    push bx
    push cx
    push dx
    push si
    push di

    cmp si, 0     ; calcula coluna a apagar em BX
    jne indo_direita

    ; indo para a esquerda -> apagar lado direito do sprite
    mov bx, cx
    add bx, 28         ; coluna a apagar = x + (largura-1)
    jmp prepara

indo_direita:
    mov bx, cx
    dec bx             ; coluna a apagar = x - 1

prepara:
    ; DI_final = DX * 320 + BX
    mov ax, dx         ; AX = y
    mov di, 320
    mul di             ; AX = y * 320
    add ax, bx         ; AX += coluna
    mov di, ax         ; DI = endereco inicial

    ; limpa verticalmente 13 pixels (escreve AL=0 e soma 320 para proxima linha)
    mov cx, 13
    xor al, al

clear_loop:
    mov es:[di], al
    add di, 320
    dec cx
    jnz clear_loop

    pop di
    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    ret
endp

inicia_video proc

    mov AL, 13h
    int 10h
    ret
endp

;-------------------Movimentacao na Tela Incial-----------------
; Funcao: Controla a animacao dos elementos na tela de menu (nave, meteoro, inimigo)
; Parametros de entrada: Variaveis de posicao dos elementos do menu
; Parametros de saida: Atualiza posicoes e redesenha elementos
;---------------------------------------------------------
movimento_tela_inicial proc
    push ax
    push bx
    push cx
    push dx
    push si
    push di
; 1) NAVE ALIADA
    ; apagar coluna antiga (direcao = direita)
    mov cx, [nave_x]
    mov dx, [nave_y]
    mov si, 1              ; 1 = direita (move_horizontal usa SI como direcao)
    call move_horizontal

    ; atualizar X
    mov ax, [nave_x]
    inc ax
    cmp ax, 320
    jl .ali_ok

    ; apagar sprite inteiro ANTES de resetar posicao
    mov cx, [nave_x]
    mov dx, [nave_y]
    mov si, OFFSET nave
    call apagar_sprite_13x29
    mov ax, 0     ; reaparece a esquerda
.ali_ok:
    mov [nave_x], ax
    ; desenhar nova posicao
    mov bx, offset nave
    mov cx, [nave_x]
    mov dx, [nave_y]
    call pinta_13X29
; 2) METEORO
    ; apagar coluna antiga (direcao = esquerda)
    mov cx, [meteoro_x]
    mov dx, [meteoro_y]
    mov si, 0              ; 0 = esquerda
    call move_horizontal

    mov ax, [meteoro_x]    ; atualizar X
    dec ax
    cmp ax, 0
    jg .met_ok

    mov cx, [meteoro_x]
    mov dx, [meteoro_y]
    mov si, OFFSET meteoro
    call apagar_sprite_13x29
    mov ax, 320          ; reaparece a direita
.met_ok:
    mov [meteoro_x], ax
    ; desenhar nova posicao
    mov bx, offset meteoro
    mov cx, [meteoro_x]
    mov dx, [meteoro_y]
    call pinta_13X29
; 3) inimigo
    ; apagar coluna antiga (usa inimigo_dir)
    mov cx, [inimigo_x]
    mov dx, [inimigo_y]
    mov si, [inimigo_dir]    ; SI = 0 (esq) ou 1 (dir)
    call move_horizontal

    ; atualizar X com ping-pong
    mov ax, [inimigo_x]
    mov bx, [inimigo_dir]    ; bx = direcao
    cmp bx, 0
    jne .inimigo_dir_right

    ; indo para esquerda
    dec ax
    cmp ax, 0
    jg .inimigo_depois
    ; bateu na esquerda -> inverter para direita
    mov word ptr [inimigo_dir], 1
    jmp .inimigo_depois
.inimigo_dir_right:
    ; indo para direita
    inc ax
    cmp ax, 291            ; limite (320 - sprite_width(29)) = 291
    jle .inimigo_depois
    ; bateu na direita -> inverter para esquerda
    mov word ptr [inimigo_dir], 0
.inimigo_depois:
    mov [inimigo_x], ax
    ; desenhar inimigo
    mov bx, offset nave_ini
    mov cx, [inimigo_x]
    mov dx, [inimigo_y]
    call pinta_13X29
    pop di
    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    ret
endp
;-------------------Movimentacao
;-------------------Gerar Inimigos------------------------------
; Funcao: Gerencia a criacao de novos inimigos baseado em tempo e limite maximo
; Parametros de entrada: [tempo_inimigo], [gen_time], [num_inimigos], [max_inimigos]
; Parametros de saida: Atualiza vetor [inimigos] e [num_inimigos]
;---------------------------------------------------------
gera_inimigo proc
    
    mov DX, gen_time            ;passa intervalo para gerar para DX
    cmp tempo_inimigo, DX     ;ve se ja deu seu tempo para estar na tela 
    jnge ainda_nao_2            ;se nao deu, ignora
    mov BX, max_inimigos      ;se deu, verifica se nao extrapolou o limite de inimigos
    cmp num_inimigos, BX      ;verifica se ja passou do limite de inimigos
    jge ainda_nao_2             ;se sim, ignora
     
    call gera_posicao_segura  ;senao, gera numero aleatorio. DX ja fica com a posicao randomica, representando sua posicao y
    inc num_inimigos          ;incrementa o numero de inimigos
    

    mov BX, offset inimigos   ;passa para BX informacoes dos inimigos

    mov AX, max_inimigos      ;passa o numero maximo de inimigos para AX
    push DX                     ;salva DX (que possui o numero aleatorio) pois ira ser perdido na multiplicacao
    mov DX, 6                   ;muda DX para 6 (tamanho do bloco de dados de cada inimigo)
    mul DX                      ;multiplica o maximo de inimigos por 6, descobrindo a posicao que estara quando tiver excedido o limite
    pop DX
    add AX, offset inimigos   ;AX carrega a posicao onde loop deve encerrar, servindo de comparador
   
testes:
    cmp WORD PTR [BX+4], 0      ;verifica se inimigo nessa posicao ja foi gerado
    jne proximo                 ;se sim, vai para o proximo no vetor
    mov WORD PTR [BX], 291      ;passa posicao da extrema direita da tela para posicao x do inimigo (320-29=291 para nao bugar)
    mov WORD PTR [BX+2], DX     ;passa o numero aleatorio para a posicao y do inimigo
    mov WORD PTR [BX+4], 1      ;liga flag de existencia do inimigo em questao
    mov tempo_inimigo, 0      ;apos criar um novo inimigo, reinicia a contagem do tempo
    jmp ainda_nao_2   
proximo:
    add BX, 6                   ;incrementa posicao do vetor em 6, indo para os dados do proximo inimigo
    cmp BX, AX                  ;verifica se ja chegou no fim do vetor
    jge ainda_nao_2             ;se sim, termina
    jmp testes                  ;senao, faz todo o processo para o novo inimigo

ainda_nao_2:
    ret
endp

;---------------------------------------------------------
; Funcao: Verifica colisao retangular entre dois objetos (29x13 pixels)
; Parametros de entrada: 
;   AX = X do objeto 1
;   BX = Y do objeto 1
;   CX = X do objeto 2
;   DX = Y do objeto 2
; Parametros de saida: DX = 1 se houve colisao, 0 caso contrario
;---------------------------------------------------------
verifica_colisao proc
    push AX             ;salva contexto
    push BX
    push CX

    ;inimigo_x + 29 > nave_x   &&   inimigo_x < nave_x + 29   &&   inimigo_y + 13 > nave_y   &&   inimigo_y < nave_y + 13
    
    add AX, 29          ;largura do inimigo (29)
    cmp AX, CX          ;compara inimigo_x com colisor_x
    jng sem_colisao
    sub AX, 29

    add CX, 29          ;largura do colisor (29)
    cmp AX, CX          ;compara inimigo_x com colisor_x
    jnl sem_colisao
    sub CX, 29
    
    add BX, 13          ;altura do inimigo (13)
    cmp BX, DX          ;compara inimigo_y com colisor_y
    jng sem_colisao
    sub BX, 13
    

    add DX, 13          ;altura do colisor (13)
    cmp BX, DX          ;compara inimigo_y com colisor_y
    jnl sem_colisao
    sub DX, 13
    
    
    mov DX, 1           ;se chegou aqui, houve colisao. Seta flag 
    jmp final_colisao   ;termina
    
sem_colisao:
    xor DX, DX          ;nao houve colisao. Desliga flag
    
final_colisao:

    pop CX          ;recupera contexto
    pop BX
    pop AX
    ret
endp
;---------------------------------------------------------
; Funcao: Atualiza posicao de todos os inimigos ativos e verifica colisoes
; Parametros de entrada: Vetor [inimigos], [nave_pos], [missil_pos]
; Parametros de saida: Atualiza posicoes, vidas, pontuacao e redesenha
;---------------------------------------------------------
move_inimigo proc
    push SI                     ;salva contexto
    push DI
    push AX
    push BX
    push CX
    push DX

    mov DI, offset inimigos   ;coloca dados do inimigo em DI

    mov AX, max_inimigos      ;passa maximo de inimigos para AX
    mov DX, 6                   ;muda DX para 6 (tamanho do bloco de dados de cada inimigo)
    mul DX                      ;multiplica o maximo de inimigos por 6, descobrindo a posicao que estara quando tiver excedido o limite
    add AX, offset inimigos   ;AX carrega a posicao onde loop deve encerrar, servindo de comparador
    mov SI, AX                  ;comparador vai para SI   


verificacoes:
    cmp WORD PTR [DI+4], 1      ;verifica se inimigo nessa posicao ja foi gerado
    jne next1                    ;se nao existe, vai para o proximo no vetor
    cmp WORD PTR [DI], 0        ;se pos_x esta no fim da tela
    jbe destroi_inimigo_s  ;destroi inimigo sem dar pontos (saiu da tela)
    
    push SI                 ; Salva contexto
    
    ; USA MOVE_HORIZONTAL PARA APAGAR RASTRO (OTIMIZACAO)
    mov CX, [DI]            ; Pega X atual
    mov DX, [DI+2]          ; Pega Y atual
    mov SI, 0               ; Direcao 0 = Esquerda (apaga rastro a direita)
    call move_horizontal
    
    ; CALCULA NOVA POSICAO
    mov ax, [inimigo_speed]
    sub WORD PTR [DI], ax   ; Subtrai a velocidade (ex: 2) do X
    
    pop SI                  ; Recupera contexto

    cmp fase, 3
    jne .pula_verificacao_terreno

    mov CX, [DI]
    mov DX, [DI+2]

    call verifica_xy_terreno_f3

    cmp ax, 1           ; Retornou 1 (Bateu)
    je .morreu_parede ; Se bateu, destroi o inimigo (sem ganhar pontos)
    jmp .pula_verificacao_terreno

.morreu_parede:
    mov CX, [DI]          ; X do inimigo
    mov DX, [DI+2]        ; Y do inimigo
    add CX, 2
    call apagar_sprite_13x29  ; garante que o desenho
    mov WORD PTR [DI+4], 0     ; Desativa a flag de existencia
    dec word ptr [num_inimigos]
    jmp next

.pula_verificacao_terreno:
    mov AX, [DI]                ;passa posicao x para AX
    mov BX, [DI+2]              ;passa posicao y para BX

    mov CX, [nave_pos]          ;passa posicao x da nave para CX
    mov DX, [nave_pos+2]        ;passa posicao y da nave para DX
    call verifica_colisao       ;verifica se houve colisao
    cmp DX, 1                   ;se sim
    je colide_player            ;trata colisao com player
    
    cmp fase, 2
    je continue

    mov CX, [missil_pos]        ;mesmo processo vericando colisao com missil
    mov DX, [missil_pos+2]
    call verifica_colisao
    cmp DX, 1
    jne continue

    cmp byte ptr [missil_exists], 1
    je destroi_inimigo_missil   ;se houver, trata (com pontos)
next1:
    jmp next
destroi_inimigo_s:
    jmp destroi_inimigo_sem_pontos   
continue:
    mov CX, [DI]             
    mov DX, [DI+2]

    cmp fase, 2
    je .desenha_meteoro

    mov BX, offset nave_ini
    jmp .pinta_agora            ;pinta inimigo com nave_ini e posicoes X e Y em CX e DX

.desenha_meteoro:
    mov BX, offset meteoro

.pinta_agora:
    call pinta_13X29
    jmp next                    ;muda para o proximo no vetor

colide_player:
    dec byte ptr [nave_vida]               ;reduz a vida da nave
    call pinta_vida             ;pinta a nova vida
    jmp destroi_inimigo_sem_pontos  ;e destroi o inimigo (sem pontos por colisao)

destroi_inimigo_missil:
    call destroi_missil         ;destroi o missil e depois o inimigo
    jmp destroi_inimigo_com_pontos  ;destroi inimigo com pontos

destroi_inimigo_com_pontos:
    mov CX, [DI]             ;apaga o inimigo em questao
    mov DX, [DI+2]

    mov SI, OFFSET nave_ini
    cmp fase, 2
    jne .apaga_pts
    mov SI, OFFSET meteoro

.apaga_pts:
    call apagar_sprite_13x29
    mov WORD PTR [DI+4], 0
    dec word ptr [num_inimigos]

    cmp fase, 3
    je .pontos_fase3

    add word ptr [pontuacao], 100  ;adiciona 100 pontos por inimigo destruido (fase 1)
    jmp .atualiza

.pontos_fase3:
    ; Fase 3
    add word ptr [pontuacao], 150

.atualiza:
    call atualiza_score  ;atualiza display do score
    call desenha_nave
    jmp next

destroi_inimigo_sem_pontos:
    mov CX, [DI]             ;apaga o inimigo em questao
    mov DX, [DI+2]

    mov SI, OFFSET nave_ini

    cmp fase, 2
    jne .apaga_sem
    mov SI, OFFSET meteoro

.apaga_sem:
    call apagar_sprite_13x29
    mov WORD PTR [DI+4], 0
    dec word ptr [num_inimigos]
    ; NAO adiciona pontos - inimigo saiu da tela
    call desenha_nave
    jmp next

destroi_inimigo:
    ; alias para manter compatibilidade - destroi com pontos
    jmp destroi_inimigo_com_pontos

next:   
    add DI, 6                   ;incrementa posicao do vetor em 6, indo para os dados do proximo inimigo         
    cmp DI, SI                  ;verifica se ja chegou no fim do vetor
    jge not_yet                 ;se sim, termina
    jmp verificacoes            ;senao, faz todo o processo para o novo inimigo

not_yet:

    pop DX                      ;recupera contexto
    pop CX
    pop BX
    pop AX
    pop DI
    pop SI
    ret
endp

;-------------------Dar tiro------------------------------------
;---------------------------------------------------------
; Funcao: Cria um missil na posicao atual da nave se nao houver um ativo
; Parametros de entrada: [nave_pos], [missil_exists]
; Parametros de saida: Define [missil_pos] e [missil_exists] = 1
;---------------------------------------------------------
gera_missil proc
    push ax
    push cx
    push dx
    
    cmp byte ptr [missil_exists], 1  ;se ja existe
    je .negado                       ;nao pode criar outro
    
    ; posicionar missil na frente da nave
    mov CX, [nave_pos]      ;posicao x da nave
    add CX, 29              ;extremo direito da nave (largura = 29)
    mov DX, [nave_pos+2]    ;posicao y da nave
    add DX, 6               ;meio da nave (altura/2 = 13/2 6)
    
    mov [missil_pos], CX    ;salva posicao x do missil
    mov [missil_pos+2], DX  ;salva posicao y do missil
    
    ; desenhar pixel branco
    mov AL, 15              ;cor branca
    call pinta_pixel
    
    mov byte ptr [missil_exists], 1  ;ativa flag de existencia
    
.negado:
    pop dx
    pop cx
    pop ax
    ret
endp

;---------------------------------------------------------
; Funcao: Atualiza a posicao do missil ativo
; Parametros de entrada: [missil_pos], [missil_speed]
; Parametros de saida: Redesenha o missil ou o desativa se sair da tela
;---------------------------------------------------------
move_missil proc
    push ax
    push cx
    push dx
    
    cmp byte ptr [missil_exists], 0  ;verifica se missil existe
    je .fim_move                     ;se nao existe, sai
    
    ; apagar pixel atual
    mov CX, [missil_pos]
    mov DX, [missil_pos+2]
    mov AL, 0                        ;cor preta (apagar)
    call pinta_pixel
    
    ; mover para direita
    mov CX, [missil_pos]
    add CX, [missil_speed]   ;velocidade do missil
    cmp CX, 319                      ;verifica limite da tela
    jg .destroy_missil               ;se passou do limite, destroi
    
    ; atualizar posicao
    mov [missil_pos], CX
    
    ; desenhar na nova posicao
    mov DX, [missil_pos+2]
    mov AL, 15                       ;cor branca
    call pinta_pixel
    jmp .fim_move
    
.destroy_missil:
    mov byte ptr [missil_exists], 0  ;desativa missil
    
.fim_move:
    pop dx
    pop cx
    pop ax
    ret
endp
;---------------Fim Mover Missil---------------

;---------------Gera Numero Aleatorio---------------
; Funcao: Gera um numero pseudo-aleatorio usando Linear Congruential Generator
; Parametros de entrada: [seed]
; Parametros de saida: DX = Numero aleatorio entre 20 e 120
;---------------------------------------------------------
gera_numero_aleatorio proc
    push ax
    push bx
    push cx
    
    mov ax, [seed]
    mov bx, 25173
    mul bx
    add ax, 13849
    mov [seed], ax
    
    ; gera numero de 20 a 120 (Area jogavel entre barra de status e terreno)
    ; - Barra de status: y = 0 a 15
    ; - area jogavel segura: y = 20 a 120  
    ; - Terreno: y = 137-199 (terrain_base_y - variacao maxima ~62)
    ; - Altura nave_ini: 13 pixels (entao y maximo = 120 para margem segura)
    xor dx, dx
    mov bx, 100  ; 120-20 = 100
    div bx
    add dx, 20   ; resultado em DX = 20 + (seed mod 100)
    
    pop cx
    pop bx
    pop ax
    ret
endp


;---------------------------------------------------------
; Funcao: Gera Y e verifica se o local na tela eh PRETO (Ceu)
; Parametros de saida: DX = Y seguro
;---------------------------------------------------------
gera_posicao_segura proc
    push ax
    push bx
    push cx
    push si
    push di
    push es

    mov ax, 0A000h
    mov es, ax

    mov cx, 15  ; Tentativas

.tentar:
    call gera_numero_aleatorio 
    push dx
    
    ; 2. Calcula endereco
    mov ax, dx
    mov bx, 320
    mul bx
    add ax, 310
    mov di, ax
    
    ; Verifica Topo
    cmp byte ptr es:[di], 0
    jne .nao_serve  ; Se falhar, vai limpar a pilha
    
    ; Verifica Pe
    add di, 4160
    cmp di, 63999
    ja .nao_serve
    cmp byte ptr es:[di], 0
    jne .nao_serve
    
    pop dx   
    jmp .fim_segura

.nao_serve:
    pop dx      ; Tira o numero ruim da pilha para nao travar o programa
    loop .tentar

    ; Se falhou todas as vezes
    mov dx, 40

.fim_segura:
    pop es
    pop di
    pop si
    pop cx
    pop bx
    pop ax
    ret
gera_posicao_segura endp

;---------------Verifica Colisao com Terreno---------------
; Funcao: Verifica se a nave colidiu com a altura do terreno na posicao X atual
; Parametros de entrada: [nave_pos]
; Parametros de saida: Decrementa vida se colidir e ajusta posicao Y
;---------------------------------------------------------
verifica_colisao_terreno proc
    push ax
    push bx
    push cx
    push dx
    push si
    push di

    cmp fase, 3
    je .modo_fase3
    
    ; calcula a posisao Y do terreno na posicao X da nave
    mov bx, [nave_pos]      ; x da nave
    
    ; usa mesma logica do render_terreno_gen para calcular altura do terreno
    mov ax, bx              ; global_x = nave_pos_x
    
    ; idx1 = global_x & 63
    mov cx, ax
    and cx, 63
    mov di, OFFSET sin_tab
    add di, cx
    mov al, [di]

    ; idx2 = (global_x >> 2) & 63
    mov cx, ax
    shr cx, 2
    and cx, 63
    mov di, OFFSET sin_tab
    add di, cx
    mov dl, [di]

    add al, dl
    shr al, 1           ; average normal
    add al, al          ; escalonamento para amplitude visivel
    mov ah, 0           ; AX = offset (0..~60)

    ; surface_y = terrain_base_y - AX
    mov cx, [terrain_base_y]
    sub cx, ax          ; CX = altura do terreno nesta posicao
    sub cx, 2
    ; verifica se a parte inferior da nave (nave_y + 13) toca o terreno
    mov ax, [nave_pos+2]    ; y da nave
    add ax, 13              ; + altura da nave
    cmp ax, cx              ; compara com altura do terreno
    jl .sem_colisao         ; se nave esta acima do terreno, ok
    
    ; colidiu com terreno - perde vida
    dec byte ptr [nave_vida]
    call pinta_vida
    
    ; reposiciona nave acima do terreno com margem de seguranca
    sub cx, 15              ; margem de seguranca
    mov [nave_pos+2], cx

.modo_fase3:
    call verifica_colisao_terreno_fase3
    
.sem_colisao:
    pop di
    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    ret
endp

;---------------------------------------------------------
; Funcao: Verifica colisao de objeto com o terreno da Fase 3
; Entrada: CX = Posicao X, DX = Posicao Y (Topo do objeto)
; Saida: AX = 1 (Bateu), 0 (Livre)
;---------------------------------------------------------
verifica_xy_terreno_f3 proc
    push bx
    push cx
    push dx
    push si
    push di
    
    ; 1. Calcular X Global
    mov ax, cx
    ;add ax, [terreno_scroll]
    
    ; 2. Descobrir indice da coluna (X / 24)
    xor dx, dx
    mov bx, 24
    div bx          ; AX = Indice Global
    
    ; 3. Wrap Around
    xor dx, dx
    mov bx, 32      ; Tamanho do mapa (TAM_MAPA_F3)
    div bx          ; DX = Resto
    
    mov si, dx      ; SI = Indice final
    
    ; 4. Ler altura do bloco
    mov bx, offset mapa_fase3_blocos
    add bx, si
    xor cx, cx
    mov cl, [bx]    ; CL = Quantidade de blocos
    
    cmp cl, 0       ; Se for 0, buraco
    je .safe
    
    ; 5. Calcular Altura Y do topo
    mov ax, cx      ; AX = Blocos
    mov bx, 16
    mul bx          
    
    mov bx, 200     ; Base 200
    sub bx, ax      ; BX = Y do Topo da Torre
    
    ; 6. Verificar se o PE do objeto toca essa altura
    mov bp, sp
    
    mov ax, [bp+4]  ; Pega o valor do "push dx" original
    ; ---------------------
    
    add ax, 13      ; Y do topo + 13 (Pe do inimigo)
    
    cmp ax, bx      ; Se Pe >= Topo_Torre, bateu
    jae .hit
    
.safe:
    pop di
    pop si
    pop dx
    pop cx
    pop bx
    mov ax, 0
    ret

.hit:
    pop di
    pop si
    pop dx
    pop cx
    pop bx
    mov ax, 1
    ret
endp

;---------------------------------------------------------
; Funcao: Verifica colissao da nave com o terreno de blocos da Fase 3
; Parametros de entrada: [nave_pos] (Coordenadas da nave)
; Parametros de saida: Atualiza [nave_vida] e ajusta [nave_pos+2] (Y) se houver coliscao
;---------------------------------------------------------
verifica_colisao_terreno_fase3 proc
    push ax
    push bx
    push cx
    push dx
    push si
    
    ; --- TESTE 1: VERIFICAR A TRASEIRA DA NAVE (X) ---
    mov ax, [nave_pos]      ; X da traseira
    call aux_verifica_ponto_f3
    cmp dx, 1               ; DX=1 significa bateu
    je .bateu
    
    ; --- TESTE 2: VERIFICAR O BICO DA NAVE (X + 28) ---
    mov ax, [nave_pos]
    add ax, 28              ; X do bico
    call aux_verifica_ponto_f3
    cmp dx, 1
    je .bateu
    
    jmp .nao_bateu

.bateu:
    dec byte ptr [nave_vida]
    call pinta_vida
    
    ; Empurra a nave para cima para nao ficar presa no bloco
    mov ax, [nave_pos+2]
    dec ax              
    mov [nave_pos+2], ax
    
.nao_bateu:
    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    ret
endp

;---------------------------------------------------------
; Funcao: Auxiliar que calcula a altura do terreno em um ponto X especifico e checa coliacso
; Parametros de entrada: AX = Coordenada X para teste, [mapa_fase3_blocos], [nave_pos]
; Parametros de saida: DX = 1 se houver colisao, 0 se estiver livre
;---------------------------------------------------------
aux_verifica_ponto_f3 proc
    push ax
    push bx
    push cx
    
    ; 1. Descobrir em qual coluna (0..13) este X estao
    ; Coluna = X / 24
    xor dx, dx      ; Zerar DX para divisao
    mov bx, 24
    div bx          ; AX = Indice da Coluna
    
    ; Protecao: Se Indice > 13, ignora (fora do mapa)
    cmp ax, 13
    ja .safe_spot
    
    ; 2. Ler quantos blocos tem nessa coluna
    mov si, offset mapa_fase3_blocos
    add si, ax      ; SI aponta para a quantidade de blocos
    xor cx, cx
    mov cl, [si]    ; CL = Quantidade de blocos (ex: 3)
    
    ; 3. Calcular a altura Y do topo da torre
    ; Formula: Topo_Y = 190 - (Blocos * 16)
    mov ax, cx      ; AX = Blocos
    mov bx, 16
    mul bx          ; AX = Altura em pixels (ex: 3 * 16 = 48)
    
    mov bx, 200     ; Base do chao
    sub bx, ax      ; BX = Y do Topo da Torre (ex: 200 - 48 = 152)
    
    ; 4. Comparar com o PE da nave
    mov ax, [nave_pos+2] ; Y da nave
    add ax, 13           ; Y do pe da nave
    
    cmp ax, bx           ; Se Pe >= Topo_Torre -> Bateu
    jae .hit_spot        ; Jump if Above or Equal
    
.safe_spot:
    pop cx
    pop bx
    pop ax
    mov dx, 0            ; Sem colisao
    ret

.hit_spot:
    pop cx
    pop bx
    pop ax
    mov dx, 1            ; Colisao!
    ret
endp

;-------------------Verificar Tecla Jogo----------------------------
; Funcao: Verifica entrada do teclado durante o jogo e move a nave ou atira
; Parametros de entrada: Buffer do teclado (INT 16h)
; Parametros de saida: Atualiza [nave_pos] ou chama gera_missil
;---------------------------------------------------------
verifica_tecla_jogo proc
    push AX
    push BX
    push CX
    push DX
    
    mov ah, 1h              
    int 16h
    jnz .tem_tecla          ; CORRECAO: Se TEM tecla, continua
    jmp .fim_verifica       ; Se NAO tem, usa JMP (que alcanca longe) para sair
.tem_tecla:
    
    xor ah,ah               
    int 16h
    
    ; --- SETA BAIXO (Down) ---
    cmp ah, 50h             
    jne .not_down
    
    cmp [nave_pos+2], 170   
    jne .executa_down       ; CORRECAO: Se NAO bateu, move
    jmp .fim_verifica       ; Se bateu, sai
.executa_down:
    ; 1. APAGA
    mov CX, [nave_pos]
    mov DX, [nave_pos+2]
    call apagar_sprite_13x29
    ; 2. MOVE
    mov ax, [nave_speed]
    add [nave_pos+2], ax
    ; 3. DESENHA
    call verifica_colisao_terreno
    call desenha_nave
    jmp .fim_verifica

.not_down: 
    ; --- SETA CIMA (Up) ---
    cmp ah, 48h             
    jne .not_up
    
    cmp [nave_pos+2], 0     
    jne .executa_up         ; CORRECAO
    jmp .fim_verifica
.executa_up:
    mov CX, [nave_pos]
    mov DX, [nave_pos+2]
    call apagar_sprite_13x29

    ; ---- LIMITE SUPERIOR DA NAVE ----
    mov ax, [nave_pos+2]   ; Y atual
    cmp ax, 16             ; topo permitido (barra vai ate ~15)
    jle .nao_sobe          ; se ja esta no limite, nao sobe

    ; sobe normalmente
    mov ax, [nave_speed]
    sub [nave_pos+2], ax
    
.nao_sobe:
    ; -------------------------------

    call verifica_colisao_terreno
    call desenha_nave
    jmp .fim_verifica

.not_up:
    ; --- SETA ESQUERDA (Left) ---
    cmp ah, 4Bh             
    jne .not_left
    
    cmp [nave_pos], 0       
    jne .executa_left       ; CORRECAO
    jmp .fim_verifica
.executa_left:
    mov CX, [nave_pos]
    mov DX, [nave_pos+2]
    call apagar_sprite_13x29
    mov ax, [nave_speed]
    sub [nave_pos], ax
    call verifica_colisao_terreno
    call desenha_nave
    jmp .fim_verifica

.not_left:
    ; --- SETA DIREITA (Right) ---
    cmp ah, 4Dh             
    jne .tiro
    
    cmp [nave_pos], 291     
    jne .executa_right
    jmp .fim_verifica
.executa_right:
    mov CX, [nave_pos]
    mov DX, [nave_pos+2]
    call apagar_sprite_13x29
    mov ax, [nave_speed]
    add [nave_pos], ax
    call verifica_colisao_terreno
    call desenha_nave
    jmp .fim_verifica

.tiro:
    ; --- ESPACO (Tiro) ---
    cmp ah, 39h             
    jne .fim_verifica
    call gera_missil        
    
.fim_verifica:    
    pop DX
    pop CX
    pop BX
    pop AX
    ret
verifica_tecla_jogo endp
;-------------------Fim Verificar Tecla Jogo----------------------------
;-------------------Verificar Tecla Menu----------------------------
; Funcao: Verifica entrada do teclado no menu (setas e enter)
; Parametros de entrada: Buffer do teclado (INT 16h)
; Parametros de saida: Atualiza [opcao_menu] ou [flag_entrada]
;---------------------------------------------------------
verifica_tecla_menu proc
    ; testa se ha tecla disponivel
    mov ah, 01h
    int 16h
    jz .sem_tecla

    ; la a tecla (bloqueante)
    mov ah, 00h
    int 16h

    ; ENTER
    cmp al, 13
    je .enter_pres

    ; verificar setas (usa AH)
    cmp ah, 48h        ; seta
    je .seta_cima

    cmp ah, 50h        ; seta
    je .seta_baixo

    jmp .sem_tecla

.seta_cima:
    ; decrementa e trava em 0
    mov ax, [opcao_menu]
    cmp ax, 0
    je .sem_tecla         ; ja no topo
    dec ax
    mov [opcao_menu], ax
    call altera_opcao
    jmp .sem_tecla

.seta_baixo:
    ; incrementa e trava em 1
    mov ax, [opcao_menu]
    cmp ax, 1
    je .sem_tecla         ; ja no fim
    inc ax
    mov [opcao_menu], ax
    call altera_opcao
    jmp .sem_tecla

.enter_pres:
    mov [flag_entrada], 1
    ret

.sem_tecla:
    ret
endp

;---------------------------------------------------------
; Funcao: Zera todo o vetor de inimigos e reseta o contador de inimigos ativos
; Parametros de entrada: Nenhum (usa offset inimigos fixo)
; Parametros de saida: Preenche vetor [inimigos] com 0 e define [num_inimigos] = 0
;---------------------------------------------------------
limpa_vetor_inimigos proc
    push ax
    push bx
    push cx
    push di
    push es

    ; Configura ES = DS para limpar variaveis
    mov ax, @data
    mov es, ax
    
    ; DI aponta para o inicio do vetor de inimigos
    mov di, OFFSET inimigos
    
    ; Tamanho: max_inimigos (10) * 6 bytes por inimigo = 60 bytes
    ; Em Words: 30 words
    mov cx, 30
    
    xor ax, ax          ; Valor 0
    cld                 ; Direcao para frente
    rep stosw           ; Preenche tudo com 0
    
    ; Zera tambem o contador
    mov word ptr [num_inimigos], 0
    
    pop es
    pop di
    pop cx
    pop bx
    pop ax
    ret
limpa_vetor_inimigos endp


;-------------------Comeca Jogo------------------------------
; Funcao: Inicializa variaveis e renderiza a tela para o inicio de uma fase
; Parametros de entrada: [tempo_fase]
; Parametros de saida: Reseta pontuacao, tempo, desenha terreno e nave
;---------------------------------------------------------
comeca_fase proc
    push ax
    push bx
    push cx
    push dx
    push bp

    call limpa_vetor_inimigos

    ;fase 3
    cmp fase, 3
    je .render_f3

    ;fase 1 e 2
    xor dx, dx
    call render_terreno_gen
    jmp .common

.render_f3:
    mov word ptr [inimigo_speed], 2
    call render_terreno_fase3

.common:
  ; Reseta posicao da nave (centro esquerda)
    mov word ptr [nave_pos], 40
    mov word ptr [nave_pos+2], 100
    xor dx,dx
    call desenha_nave
    
    ; inicializa temporizador
    mov ax, [tempo_fase]
    xor dx, dx
    mov bx, 60
    div bx
    mov [tempo_segundos], ax
    mov byte ptr [frame_counter], 0
    ; inicializa ticks BIOS tracking
    mov ah, 00h
    int 1Ah
    mov [bios_last_tick], dx
    mov byte ptr [bios_tick_acc], 0

    pop bp
    pop dx
    pop cx
    pop bx
    pop ax
    ret
    
endp

    mov byte ptr [vidas], 3
    
    ; Reseta Score e Tempo
    mov word ptr [pontuacao], 0
    mov word ptr [tempo_segundos], 0
    mov byte ptr [frame_counter], 0
    
    ; Reseta flags de controle
    mov byte ptr [missil_exists], 0
    mov word ptr [flag_entrada], 0
    mov word ptr [num_inimigos], 0
    mov word ptr [tempo_inimigo], 0
    
    ; Limpa vetor de inimigos (garantia extra)
    call limpa_vetor_inimigos

    mov word ptr [inimigo_speed], 1
    
    ; Reseta Posicoes Iniciais (Menu)
    mov word ptr [nave_x], 10
    mov word ptr [nave_y], 50
    mov word ptr [meteoro_x], 300
    
    ; Reseta Posicao do Jogador para o jogo
    mov word ptr [nave_pos], 160
    mov word ptr [nave_pos+2], 100

    pop es
    pop ds
    pop di
    pop si
    pop cx
    pop ax
    ret
endp


;-------------------Inicio-----------------------------------------
INICIO:
    INICIO:
    MOV AX, @data
    MOV DS, AX
    MOV AX, 0A000h
    mov ES, AX
    
    call resetar_jogo
    
    ; gerar posicao X aleatoria
    call gerar_posicao_inimigo

    call inicia_video

    ; DESENHA O MENU COMPLETO AQUI
    call desenha_menu_inicial

menu_loop:

    call movimento_tela_inicial

    mov cx, [game_delay]
delay_lp:
    loop delay_lp

    call verifica_tecla_menu

    cmp flag_entrada, 1
    jne .apenas_mover

    mov flag_entrada, 0

    cmp opcao_menu, 0
    je jogo

    cmp opcao_menu, 1
    je sair
    jmp menu_loop

.apenas_mover:
    call altera_opcao
    jmp menu_loop

jogo:
    inc fase
    call verificar_fase2  ; configurar tabelas e cores para a fase atual
    call mudar_tela

    cmp fase, 1
    je .d1
    cmp fase, 2
    je .d2
    cmp fase, 3
    je .d3
    jmp fim

.d1:
    call desenha_fase1
    jmp continua
.d2:
    call desenha_fase2
    jmp continua
.d3:
    call desenha_fase3

continua:
    ; espera 4 segundos na tela da fase
    mov al, 4
    call delay_segundos

    ; volta para tela de jogo limpa e desenha barra de status
    call mudar_tela
    call comeca_fase
    call desenha_status_bar

loop_jogo:
    ; checa ESC para sair
    call verifica_tecla_jogo
    mov ah, 01h
    int 16h
    jz .no_key2
    mov ah, 00h
    int 16h
    cmp al, 27
    je sair
.no_key2:

    ; atraso de frame simples
    mov cx, [game_delay]
._delay:
    loop ._delay

    call durante_jogo

    cmp byte ptr [nave_vida], 0
    je sair_game_over   ; Se vida = 0, vai para Game Over

    mov ax, [tempo_segundos]
    cmp ax, 0
    jne loop_jogo       ; Se tem tempo, continua jogando
    
    cmp fase, 3
    je ganhou_jogo      ; Se acabou tempo na Fase 3 -> VITORIA
    
    jmp jogo            ; Se acabou tempo na Fase 1 ou 2 -> PROXIMA FASE

sair:
    jmp fim

; --- GAME OVER---
sair_game_over:
    call mudar_tela
    call desenha_game_over
    
    ; Espera tecla para voltar ao menu
    mov ah, 00h
    int 16h
    
    jmp INICIO          ; Volta para o Menu Principal

; --- VITORIA ---
ganhou_jogo:
    call mudar_tela
    call desenha_vitoria
    
    ; Espera tecla para voltar ao menu
    mov ah, 00h
    int 16h
    
    jmp INICIO          ; Volta para o Menu Principal

fim:
    mov ah, 4Ch     
    mov al, 0      
    int 21h

end INICIO
;-------------------Fim Inicio-----------------------------------------
